==================================================
1. LINUX VST BUILD AND INSTALL
==================================================

Build all four current VST variants:

cd /home/north3rnlight3r/Projects/AifredVst-2.2.4-beta

cmake -S . -B build/baseline-vst-release \
  -G Ninja \
  -DCMAKE_BUILD_TYPE=Release \
  -DDAWAI_BUILD_APP=OFF \
  -DDAWAI_ENABLE_TESTS=OFF \
  -DDAWAI_BUILD_AIFRED_PLUGIN=ON

cmake --build build/baseline-vst-release \
  --target aifred_vst3_A aifred_vst3_B aifred_vst3_C aifred_vst3_D \
  --parallel 8

cmake --build build/baseline-vst-release \
  --target aifred_vst3_A_VST3 aifred_vst3_B_VST3 aifred_vst3_C_VST3 aifred_vst3_D_VST3 \
  --parallel 8

Install the built VST variants to the Linux user VST3 folder:

mkdir -p "$HOME/.vst3"

cp -a build/baseline-vst-release/Source/aifred_vst3_A_artefacts/Release/VST3/AIFRED_a.vst3 "$HOME/.vst3/AIFRED_v2.2.4a.vst3"
cp -a build/baseline-vst-release/Source/aifred_vst3_B_artefacts/Release/VST3/AIFRED_b.vst3 "$HOME/.vst3/AIFRED_v2.2.4b.vst3"
cp -a build/baseline-vst-release/Source/aifred_vst3_C_artefacts/Release/VST3/AIFRED_c.vst3 "$HOME/.vst3/AIFRED_v2.2.4c.vst3"
cp -a build/baseline-vst-release/Source/aifred_vst3_D_artefacts/Release/VST3/AIFRED_d.vst3 "$HOME/.vst3/AIFRED_v2.2.4d.vst3"
